PostgreSQL Trigger 1 SQL Code 
CREATE OR REPLACE FUNCTION check_stock_before_order()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.quantity > (SELECT stock_quantity FROM store_product WHERE product_id = NEW.product_id) THEN
        RAISE EXCEPTION 'Insufficient stock! Requested % units exceeds available stock.', NEW.quantity;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_stock
BEFORE INSERT ON store_orderdetail
FOR EACH ROW EXECUTE FUNCTION check_stock_before_order();

PostgreSQL Trigger 2 SQL Code 
CREATE OR REPLACE FUNCTION calculate_final_payment()
RETURNS TRIGGER AS $$
BEGIN
    NEW.final_amount := NEW.amount + NEW.tax - NEW.discount;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_auto_payment_calc
BEFORE INSERT OR UPDATE ON store_payment
FOR EACH ROW EXECUTE FUNCTION calculate_final_payment();

PostgreSQL Trigger 3 SQL Code 
CREATE OR REPLACE FUNCTION audit_deleted_order()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO store_orderlog (order_id, deletion_timestamp, details)
    VALUES (OLD.order_id, NOW(), CONCAT('AUDIT LOG: Order #', OLD.order_id, ' deleted. Total: $', OLD.total_amount));
    RETURN OLD;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_audit_order_deletion
AFTER DELETE ON store_order
FOR EACH ROW EXECUTE FUNCTION audit_deleted_order();
